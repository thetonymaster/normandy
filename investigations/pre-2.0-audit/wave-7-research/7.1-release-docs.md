### 1.6 Scripts (verbatim, rerunnable)

`api_dump.exs`. Usage: `elixir api_dump.exs <ebin>`.

```elixir
[ebin] = System.argv()
for beam <- Path.wildcard(Path.join(ebin, "*.beam")) |> Enum.sort() do
  mod = beam |> Path.basename(".beam")
  case Code.fetch_docs(beam) do
    {:docs_v1, _, _, _, moddoc, _, entries} ->
      mhidden = moddoc == :hidden
      IO.puts("M #{mod}#{if mhidden, do: " hidden", else: ""}")
      for {{kind, name, arity0}, _, _, doc, meta} <- entries,
          kind in [:function, :macro, :callback, :macrocallback, :type],
          arity <- (arity0 - Map.get(meta, :defaults, 0))..arity0 do
        tag = case doc do :hidden -> " hidden"; :none -> " nodoc"; _ -> "" end
        IO.puts("F #{mod}.#{name}/#{arity} #{kind}#{tag}#{if mhidden, do: " modhidden", else: ""}")
      end
    {:error, reason} -> IO.puts("E #{mod} #{inspect(reason)}")
  end
end
```

The public sets come from these awk lines:

```sh
awk '$1=="M" && $3!="hidden"{print $2}' api_X.txt | sort > mods_X.txt
awk '$1=="F" && $4!="hidden" && $4!="modhidden" && $5!="modhidden" && $3!="type"{print $2" "$3}' api_X.txt | sort -u > funs_X.txt
comm -23 funs_main.txt funs_head.txt   # removed or re-arity'd
```

`spec_dump.exs` prints `Mod.f/a<TAB>spec` for documented functions. It uses `Code.Typespec.fetch_specs(File.read!(beam))` (a beam binary, not a path) and `Code.Typespec.spec_to_quoted/2 |> Macro.to_string`. The diff is `join -t $'\t' spec_main.txt spec_head.txt | awk -F'\t' '$2!=$3'`.

`struct_dump.exs` runs `Code.prepend_path(ebin)`; for each non-hidden module exporting `__struct__/0` it prints the sorted keys.

Per-file option-key check (zsh: use `while read`, since `for k in $var` does not word-split):

```sh
git ls-tree -r --name-only main lib | grep '\.ex$' | while read f; do
  git show main:$f | grep -oE "(Keyword\.(get|fetch!?|has_key\?|pop|get_lazy)\((opts|options)[a-z_]*, :[a-z_?!]+|(opts|options)\[:[a-z_?!]+\])" \
   | grep -oE ":[a-z_?!]+$|:[a-z_?!]+\]" | tr -d ']' | sort -u | while read k; do n=${k#:}
     if [ ! -f "$f" ]; then echo "GONE-FILE $f $k"; elif ! grep -qE "(\b$n:|:$n\b)" $f; then echo "$f $k"; fi; done; done
```

```elixir
defmodule Normandy.DocsGroupsTest do
  # Every documented module in lib/ is in exactly one mix.exs docs group, and every
  # grouped name is a documented module. Protocol implementations are skipped: ExDoc
  # does not list them.
  use ExUnit.Case, async: true

  test "every documented lib/ module is in exactly one docs group" do
    groups = Mix.Project.config() |> Keyword.fetch!(:docs) |> Keyword.fetch!(:groups_for_modules)

    grouped =
      for {group, mods} <- groups, mod <- mods do
        assert is_atom(mod), "group #{inspect(group)} lists #{inspect(mod)}; use module atoms only"
        {mod, group}
      end

    {:ok, modules} = :application.get_key(:normandy, :modules)

    documented =
      for mod <- modules,
          Code.ensure_loaded!(mod),
          lib_module?(mod),
          not function_exported?(mod, :__impl__, 1),
          match?({:docs_v1, _, _, _, doc, _, _} when doc != :hidden, Code.fetch_docs(mod)),
          do: mod

    counts = Enum.frequencies_by(grouped, &elem(&1, 0))
    ungrouped = documented |> Enum.reject(&Map.has_key?(counts, &1)) |> Enum.sort()
    duplicated = for {mod, n} <- counts, n > 1, do: mod
    unknown = Map.keys(counts) -- documented

    assert ungrouped == [], "documented but in no docs group: #{inspect(ungrouped)}"
    assert duplicated == [], "in more than one docs group: #{inspect(duplicated)}"
    assert unknown == [], "grouped but hidden, a protocol impl or missing: #{inspect(unknown)}"
  end

  defp lib_module?(mod) do
    source = mod.module_info(:compile) |> Keyword.fetch!(:source) |> List.to_string()
    String.starts_with?(Path.relative_to_cwd(source), "lib/")
  end
end
```

Gotchas I hit and fixed:
- In the test env, `:application.get_key(:normandy, :modules)` includes `test/support` modules (contracts, mocks). Hence `lib_module?/1`.
- Modules are not loaded yet, so `function_exported?(_, :__impl__, 1)` is false until `Code.ensure_loaded!/1`.
- Protocols are not consolidated in test (`mix.exs:14`), which does not matter here.
